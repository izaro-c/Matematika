  <Ordenacion

    pregunta="Arrastatu urratsak dedukzio-ordena egokian jartzeko:"
    pasos={[
      "$a^2 + 8^2 = 10^2$",
      "$a^2 + 64 = 100$",
      "$a^2 = 100 - 64$",
      "$a = \\sqrt{36} = 6$"
    ]}
  >
    <Ordenacion.ErrorComun titulo="Erro karratua kenketa sinplifikatu aurretik ateratzea">
      Ezin da erroa gai bakoitzari banan-banan aplikatu ($\sqrt{100 - 64} \neq \sqrt{100} - \sqrt{64} = 10 - 8 = 2$). Lehenik $100 - 64 = 36$ kalkulatu behar da, eta ondoren erro karratua atera.
    </Ordenacion.ErrorComun>

    <Ordenacion.Resolucion>
      Dedukzio-ordena zuzena hauxe da:
      1. Planteamendua: $a^2 + 8^2 = 10^2$
      2. Potentziak: $a^2 + 64 = 100$
      3. Bakantzea: $a^2 = 100 - 64 = 36$
      4. Erro karratua: $a = \sqrt{36} = 6$
    </Ordenacion.Resolucion>
  </Ordenacion>
</ExerciseStep>
