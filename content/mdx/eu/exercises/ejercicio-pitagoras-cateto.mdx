      </Formula>
    </Emparejar.Resolucion>

    pregunta="Arrastatu urratsak ordena deduktibo egokian jartzeko:"
    pasos={[
      "$a^2 + 8^2 = 10^2$",
      "$a^2 + 64 = 100$",
      "$a^2 = 100 - 64$",
      "$a = \\sqrt{36} = 6$"
    ]}
  >
    <Ordenacion.ErrorComun titulo="Erro karratua kenketa sinplifikatu aurretik ateratzea">
      Ezin da erro karratua gaiz gai aplikatu ($\sqrt{100 - 64} \neq \sqrt{100} - \sqrt{64} = 10 - 8 = 2$). Lehenik $100 - 64 = 36$ kendura kalkulatu behar da eta ondoren erroa atera.
    </Ordenacion.ErrorComun>

    <Ordenacion.Resolucion>
      Urrats deduktibo zuzenak hauek dira:
      - Planteamendua: $a^2 + 8^2 = 10^2$
      - Berreturen kalkulua: $a^2 + 64 = 100$
      - Ezezagun koadratikoaren bakantzea: $a^2 = 100 - 64 = 36$
      - Erro karratuaren kalkulua: $a = \sqrt{36} = 6$
    </Ordenacion.Resolucion>
  </Ordenacion>
</ExerciseStep>
