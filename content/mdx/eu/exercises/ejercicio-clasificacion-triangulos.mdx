    <Clasificador.ErrorComun titulo="Hirukote pitagorikoa edo angelu kamutsak nahastea">
      Gogoratu <ConceptLink targetId="teorema-pitagoras">Pitagorasen teoremaren</ConceptLink> bidez ($3^2 + 4^2 = 5^2$) 3, 4 eta 5 aldeek triangelu angeluzuzena osatzen dutela. Era berean, 90° baino handiagoa den angelu bakar batek triangelua kamuts bihurtzen du.
    </Clasificador.ErrorComun>

    <Clasificador.Resolucion>
      - **Angeluzuzenak**: 90°-ko angelu bat dute ($90^\circ, 45^\circ, 45^\circ$ edo $3, 4, 5$ hirukotea).
      - **Zorrotzak**: Beren 3 angeluak 90° baino zorrozki txikiagoak dira ($60^\circ, 60^\circ, 60^\circ$).
      - **Kamutsak**: 90° baino zorrozki handiagoa den angelu bat dute ($120^\circ, 30^\circ, 30^\circ$).
    </Clasificador.Resolucion>
  </Clasificador>

    correct={[["2", "1", "1"]]}
  >
    <MatrizInteractiva.ErrorComun titulo="Elementuen zenbaketa okerra">
      Berrikusi aurreko urratseko elementuak: 2 triangelu angeluzuzen daude ($90^\circ, 45^\circ, 45^\circ$ eta $3, 4, 5$), zorroztasun bat ($60^\circ, 60^\circ, 60^\circ$) eta kamuts bat ($120^\circ, 30^\circ, 30^\circ$).
    </MatrizInteractiva.ErrorComun>

    <MatrizInteractiva.Resolucion>
      Sailkatutako 4 triangeluak biltzean 2 angeluzuzen, 1 zorrotz eta 1 kamuts lortzen dira. Dagokion errenkada-matrizea $[2 \quad 1 \quad 1]$ da.
    </MatrizInteractiva.Resolucion>
  </MatrizInteractiva>
